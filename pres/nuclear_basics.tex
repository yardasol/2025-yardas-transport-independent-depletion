\begin{frame}
    \frametitle{Depletion}
        \begin{columns}
            \column[t]{5cm}
               Concentration of nuclides in a reactor are constantly changing
               due to fission, neutron capture, and decay. 
               \newline
               \newline
               Why do we care?
               \begin{itemize}
                   \item Strong coupling between depletion and transport
                   \begin{itemize}
                       \item Neutron poisions $\rightarrow$ affects neutron economy
                       \item $\downarrow$ U, $\uparrow$ Pu $\rightarrow$ Change
                           in fission neutron energy distribution
                   \end{itemize}
                   \item Fission product decay heat $\rightarrow$ affects fuel
                       reprocessing/disposal
               \end{itemize}
            \column[t]{5cm}
                \begin{figure}[htbp!]
                    \begin{center}
                        \includegraphics[height=5cm]{./images/isotopes_over_time.png}
                    \end{center}
                    \caption{Buildup of plutionium isotopes with a burnup for a
                    typical LWR fuel composition. Reproduced from Figure 6-2 in
                    \cite{knief_nuclear_1981}.}
                    \label{fig:isotopes-over-time}
                \end{figure}
        \end{columns}
\end{frame}
